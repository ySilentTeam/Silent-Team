  local jy = jj(jt, os.clock())
  local jz = (not jw or jn(jt, jx))
  local ka = (jw and ((jz and (jy > is))))
  local kb = ((jw and (((jz and is) or it))) or is)
  if ka then
    kb = it
  end
  local kc = math.max(0, (kb - jy))
  local kd = math.min(math.max((jv or 0), 0), (kc / jx))
  local ke = math.max(0, (kc - (kd * jx)))
  if ka then
    ke = 0
  end
  return jx, kd, ke, jy, jz
end
local kf, kg, kh, ki = 4, 8, 105, 65
local kj, kk, kl = 0.05, 0.6, 14
local km, kn, ko = 20, 14, 0.55
local kp = Vector3.zero
local kq, kr, ks, kt = 0, 0, 4, 0
local function ku(kv)
  if not af.spoofWalkEnabled then
    kt = 0
    return
  end
  if ((af.flyEnabled or af.autoFarmPescaEnabled) or af.farmPartesEnabled) then
    kt = 0
    return
  end
  local kw = ie()
  if not kw then
    return
  end
  local kx = ic()
  if not kx then
    return
  end
  if (kx.Sit or iq[kx:GetState()]) then
    kt = 0
    return
  end
  if (ih.Magnitude <= 0) then
    kt = 0
    if ((os.clock() < kq) and (kp.Magnitude > 0)) then
      local ky = math.max(kx.WalkSpeed, 1)
      local kz = math.clamp((kn / ky), 0, 1)
      pcall(kx.Move, kx, (kp * kz), false)
    end
    return
  end
  kp = ih.Unit
  kq = (os.clock() + ko)
  local la = kw.AssemblyLinearVelocity
  local lb = Vector3.new(la.X, 0, la.Z).Magnitude
  if (((((kv or 0)) >= kj) or kw.Anchored) or (lb < 12)) then
    kr = (os.clock() + kk)
  end
  local lc = ik()
  local ld = math.min(af.spoofWalkSpeed, ((lc and af.spoofWalkSpeed) or ((((os.clock() < kr) and ki) or kh))))
  local le = (ld - ij)
  if (le <= 0) then
    kt = 0
    return
  end
  local lf = (0.9 + (math.random() * 0.1))
  local lg = ((((le * kv)) * lf) + kt)
  local lh = math.min(lg, (le * kv), ((lc and kg) or kf))
  kt = math.clamp((lg - lh), 0, ks)
  local li = ih.Unit
  kw.CFrame = (kw.CFrame + (li * lh))
  local lj = kw.AssemblyLinearVelocity
  if (Vector3.new(lj.X, 0, lj.Z).Magnitude < kl) then
    kw.AssemblyLinearVelocity = Vector3.new((li.X * km), lj.Y, (li.Z * km))
  end
end

end

local function mo(mp)
  local mq = ie()

  if (not mq or not mr) then
    return
  end
  if not af.flyEnabled then
    mr:SetStateEnabled(Enum.HumanoidStateType.Swimming, true)
    mr:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
    if (mr:GetState() == Enum.HumanoidStateType.Swimming) then
      local ms = (((mr.FloorMaterial == Enum.Material.Air) and Enum.HumanoidStateType.Freefall) or Enum.HumanoidStateType.Running)
      pcall(mr.ChangeState, mr, ms)
    end
    if (af.autoFarmPescaEnabled or af.farmPartesEnabled) then
      mn = false
      return
    end
    if ik() then
      mn = false
      mq.AssemblyLinearVelocity = Vector3.zero
      return
    end
    local mt, mu, mv, mw, mx = js(mq, mp, mk, true)
    if (mn and ((mx and (mm.Magnitude > 0)))) then
      mq.AssemblyLinearVelocity = (mm * mu)
      return
    end
    mn = false
    mq.AssemblyLinearVelocity = Vector3.zero
    return
  end
  if (mr.Health <= 0) then
    return
  end
  mr:SetStateEnabled(Enum.HumanoidStateType.Jumping, false)
  mr:ChangeState(Enum.HumanoidStateType.Swimming)
  local my = 0
  if b:IsKeyDown(Enum.KeyCode.Space) then
    my = 1
  elseif b:IsKeyDown(Enum.KeyCode.LeftControl) then
    my = -1
  end
  local mz = (mr.MoveDirection + Vector3.new(0, my, 0))
  if (mz.Magnitude <= 0) then
    if ik() then
      mq.AssemblyLinearVelocity = Vector3.zero
      return
    end
    local na, nb, nc, nd, ne = js(mq, mp, mk, true)
    if (ne and (mm.Magnitude > 0)) then
      mq.AssemblyLinearVelocity = (mm * nb)
    else
      mq.AssemblyLinearVelocity = Vector3.zero
    end
    return
  end
  mz = mz.Unit
  mm = mz
  mn = true
  if ik() then
    mq.AssemblyLinearVelocity = (mz * af.flySpeed)
    return
  end
  local nf = (((mr.MoveDirection.Magnitude > 0) and mi) or ml)
  local ng = math.min(af.flySpeed, nf)
  local nh = math.min(ng, mj)
  local ni, nj, nk, nl, nm = js(mq, mp, nh, true)
  local nn = math.min(ng, ((nm and iu) or mi))
  local no = math.max(0, (nn - nj))
  if (no > 0) then
    mq.CFrame = (mq.CFrame + ((mz * no) * ((mp or 0.016))))
  end
  mq.AssemblyLinearVelocity = (mz * nj)
end

      end
    end)
  end

  if not asw.dragging then
    return
  end
  if (asy.UserInputType == Enum.UserInputType.MouseMovement) then
    local asz = (asy.Position - asw.startMousePos)
    ajo.Position = UDim2.new(asw.startPos.X.Scale, (asw.startPos.X.Offset + asz.X), asw.startPos.Y.Scale, (asw.startPos.Y.Offset + asz.Y))
  elseif (asy.UserInputType == Enum.UserInputType.Touch) then
    if (asw.touchInput and (asy ~= asw.touchInput)) then
      return
    end
    local ata = (asy.Position - asw.startMousePos)
    ajo.Position = UDim2.new(asw.startPos.X.Scale, (asw.startPos.X.Offset + ata.X), asw.startPos.Y.Scale, (asw.startPos.Y.Offset + ata.Y))
  end
end)

  if ((atb.UserInputType == Enum.UserInputType.MouseButton1) or (atb.UserInputType == Enum.UserInputType.Touch)) then
    asw.dragging = false
    asw.touchInput = nil
  end
end)
